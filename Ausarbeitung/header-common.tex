% ──────────────────────────────────────────────────────────────────────────────
% Encoding and Language
% ──────────────────────────────────────────────────────────────────────────────

% Input encoding: UTF-8 (required for non-ASCII characters)
\usepackage[utf8]{inputenc}

% Output encoding: T1 (required for proper hyphenation of non-ASCII words)
\usepackage[T1]{fontenc}

% Language: German (hyphenation, date format, etc.)
\usepackage[ngerman]{babel}

% ──────────────────────────────────────────────────────────────────────────────
% Fonts and Typography
% ──────────────────────────────────────────────────────────────────────────────

% Latin Modern: better font with wider character coverage
\usepackage{lmodern}

% Inconsolata: nicer monospace font for code listings
\usepackage{inconsolata}

% Non-breaking spaces and unicode character support
\usepackage{newunicodechar}

% ──────────────────────────────────────────────────────────────────────────────
% Mathematics
% ──────────────────────────────────────────────────────────────────────────────

% AMS math environments and macros
\usepackage{amsmath}

% AMS theorem environments
\usepackage{amsthm}

% AMS extra symbols
\usepackage{amssymb}

% AMS fonts (blackboard bold, etc.)
\usepackage{amsfonts}

% Mathtools: improvements and extensions to amsmath
\usepackage[centerdots]{mathtools}

% Double-struck fonts (indicator functions, etc.)
\usepackage{dsfont}

% IPA phonetic symbols and other text symbols
\usepackage{tipa}

% Nice-looking fractions in text mode
\usepackage{nicefrac}

% ──────────────────────────────────────────────────────────────────────────────
% Graphics and Diagrams
% ──────────────────────────────────────────────────────────────────────────────

% Core graphics package
\usepackage{graphicx}

% TikZ: programmable graphics
\usepackage{tikz}

% TikZ libraries
\usetikzlibrary{chains,shapes.multipart}
\usetikzlibrary{shapes,arrows,positioning}
\usetikzlibrary{positioning,fit,shapes.geometric}

% Picture extensions
\usepackage{pict2e}

% Side-by-side figures and sub-captions
\usepackage{subcaption}

% ──────────────────────────────────────────────────────────────────────────────
% Colors
% ──────────────────────────────────────────────────────────────────────────────

% Color support (xcolor loads colortbl as well)
\usepackage{xcolor}
\usepackage{colortbl}

% ──────────────────────────────────────────────────────────────────────────────
% Tables and Floats
% ──────────────────────────────────────────────────────────────────────────────

% Float placement options
\usepackage{float}

% Professional-looking tables (toprule, midrule, bottomrule)
\usepackage{booktabs}

% Array enhancements
\usepackage{array}

% ──────────────────────────────────────────────────────────────────────────────
% Code Listings
% ──────────────────────────────────────────────────────────────────────────────

% Code listings with syntax highlighting
\usepackage{listings}

% ──────────────────────────────────────────────────────────────────────────────
% Miscellaneous

% Colorful text boxes
\usepackage{tcolorbox}

% Blind text for testing layouts

% Arrow and symbol enhancements
\usepackage{stmaryrd}

% LaTeX standard symbols
\usepackage{latexsym}

% Todo notes for draft mode
\usepackage{todonotes}

% Customizable list environments
\usepackage{enumitem}

% (Experimental) Dynamic image scaling
\newcommand{\dynamicscale}[1]{%
	\includegraphics[width=\minof{\textwidth}{\widthof{\includegraphics{#1}}-\parindent}]{#1}%
}

% (Experimental) Dynamic TikZ scaling
\newcommand{\dynamictikz}[1]{%
	\begin{tikzpicture}[scale=1]
		\draw (0,0) rectangle (\minof{\textwidth}{\widthof{#1}}-\parindent, \heightof{#1});
		#1
	\end{tikzpicture}%
}

% Define a 4th level heading: \subsubsubsection
\makeatletter
\newcommand\subsubsubsection{\@startsection{subsubsubsection}{4}{\z@}%
	{-3.25ex \@plus -1ex \@minus -.2ex}%
	{1.5ex \@plus .2ex}%
	{\normalfont\normalsize\bfseries}}
\newcounter{subsubsubsection}[subsubsection]
\renewcommand\thesubsubsubsection{\thesubsubsection.\@arabic\c@subsubsubsection}
\newcommand*\l@subsubsubsection{\@dottedtocline{3}{10.0em}{4.1em}}
\newcommand*{\subsubsubsectionmark}[1]{}
\makeatother

\setcounter{secnumdepth}{4}
\setcounter{tocdepth}{4}

		False,

		set,
		repeat,
		break,
		continue,
		Eingabe,
		Ausgabe,
		return,
		print,
		exit
	},
	sensitive=false,
	delim=[l][keywordstyle]{:},
	comment=[l][commentstyle]{\#}
}
