% Separator slide with large centered text
% Usage: \trennfolie{Section Title}
\newcommand{\trennfolie}[1]{
	\begin{frame}[plain]
		\centering
		\vfill
		\Huge \textbf{#1}
		\vfill
	\end{frame}
}
