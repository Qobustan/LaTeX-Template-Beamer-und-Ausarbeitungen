% ==============================================================================
% header-common.tex
% Gemeinsame Pakete für Vortrag und Ausarbeitung
% Common packages for presentation and paper
%
% ARCHITEKTUR-HINWEIS: Diese Datei enthält NUR \usepackage-Aufrufe, die sowohl
% mit dem Beamer-Dokumentenstil (Vortrag) als auch mit scrartcl (Ausarbeitung)
% kompatibel sind. Konfigurationsbefehle (\definecolor, \hypersetup, \setlist,
% \newtheorem, \tikzset, \newcommand usw.) stehen in den dokumenttyp-spezifischen
% Headern (header-article.tex, header-beamer.tex), weil Beamer viele dieser
% Befehle intern bereits setzt und Dopplungen zu Konflikten führen.
%
% BEWUSST NICHT IN DIESER DATEI:
%   - inputenc / fontenc  → Engine-spezifisch; in Ausarbeitung.tex / Vortrag.tex
%                           per \ifluatex-Konditional gesetzt
%   - babel               → Sprachauswahl ist dokumentspezifisch; in
%                           Ausarbeitung.tex / Vortrag.tex mit Hinweis zur
%                           Konfiguration (z.B. ngerman → english)
%   - amsthm              → Inkompatibel mit Beamers eigenem Theorem-System;
%                           nur in header-article.tex geladen
%   - tipa                → Wurde entfernt: Das Paket liefert phonetische IPA-
%                           Symbole und definiert viele Text-Befehle um, was zu
%                           Konflikten mit anderen Schriftpaketen führt. In
%                           diesem Template wird tipa an keiner Stelle verwendet.
%   - natbib / cite       → Inkompatibel mit biblatex (dem hier genutzten
%                           Literaturpaket); war auskommentierter Altcode
%
% ==============================================================================

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

% Für LuaLaTeX/pdfLaTeX Unterscheidung
\usepackage{iftex}

% Bildoptionen
\usepackage{pict2e}

% Latex-Symbole
\usepackage{latexsym}

% Zusätzliche Kodierungen (nur für pdflatex; LuaLaTeX nutzt fontspec)
\ifluatex\else
  \usepackage{inconsolata}
\fi

% Nicht-umbrechbare Leerzeichen
\usepackage{newunicodechar}

% Abbildungen
\usepackage{tikz}
\usepackage{pgfplots}
\pgfplotsset{compat=1.18}

% Weitere TikZ-Erweiterungen
\usetikzlibrary{chains,shapes.multipart}

% Erweiterungen für die Abbildungen mit TikZ
\usetikzlibrary{shapes,arrows,positioning}

% Abbildungen nebeneinander stellen und Text anordnen
\usepackage{subcaption}

% Mathematische Symbole: dsfont (Doppelstrich-Schrift), amssymb (zusätzliche Symbole)
\usepackage{dsfont}
\usepackage{amssymb}

% Schöner Aufzählungspunkt; und andere schöne Textelemente
\usepackage{textcomp}

% Pfeiloptimierung; und ähnliches
\usepackage{stmaryrd}

% Blindtext (zur Hilfe beim Schreiben)
\usepackage{blindtext}

% Text-Boxen
\usepackage{tcolorbox}

% Grafiken
\usepackage{graphicx}

% Für korrekte Anführungszeichen mit biblatex (behebt Warning)
\usepackage{csquotes}

% Farbiger Text
\usepackage{xcolor}
\usepackage{colortbl}

% Erweiterungen für den Mathemodus
\usepackage{amsmath}

% Erweiterung: Mathefonts
\usepackage{amsfonts}

% Noch mehr Mathe
\usepackage[centerdots]{mathtools}

% Für Code/Pseudocode
\usepackage{listings}

% Eventuell erforderlich
\usepackage{array}

% Klickbare Verweise
\usepackage{hyperref}

% Schönere Schriftart (nur für pdflatex; LuaLaTeX nutzt fontspec)
\ifluatex\else
  \usepackage{lmodern}
\fi

% Tabellen ausrichten
\usepackage{float}

% Um schönere Brüche darstellen zu können
\usepackage{nicefrac}

% Todo-Anmerkungen einfügen
\usepackage{todonotes}
\usepackage{enumitem}
\usepackage{booktabs}

% Algorithmen in Pseudocode
\usepackage{algorithm}
\usepackage{algpseudocode}
